{{- /* Photo wall: shows every image in this page's `photos/` folder. Add a photo: drop a .jpg/.png/.webp into content/gallery/photos/. Newest first: files are sorted by name, descending, so start names with the date, e.g. 2026-11-agu-talk.jpg. Caption (optional): add an entry under `photos:` in the page's front matter: photos: - file: 2026-11-agu-talk.jpg caption: Invited talk at AGU 2026 Without one, the caption is made from the file name. */ -}} {{- $captions := dict -}} {{- range .Page.Params.photos -}} {{- $captions = merge $captions (dict .file .caption) -}} {{- end -}} {{- $imgs := (.Page.Resources.Match "photos/*").ByType "image" -}} {{- $imgs = sort $imgs "Name" "desc" -}}
{{- range $imgs -}} {{- $file := path.Base .Name -}} {{- $caption := index $captions $file -}} {{- if not $caption -}} {{- $caption = replaceRE `^\d{4}(-\d{2})?(-\d{2})?-` "" (strings.TrimSuffix (path.Ext $file) $file) | replaceRE `[-_]+` " " | humanize -}} {{- end -}} {{- $thumb := . -}} {{- if gt .Width 900 }}{{ $thumb = .Resize "900x webp q85" }}{{ end -}} {{- end -}}
